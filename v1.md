- Farcaster project board using Neynar score voting (FREE, no tokens)
- Basic feature submission & voting
- Status management (open/in progress/shipped/hidden)

- Bot with multi-project routing & auto-extraction
- New projects use token voting (1% fee)
- Free project creation via bot tagging
- Auto-tagging & similarity merging
- Sub-items with independent voting
- Project admin settings
- Embeddable widget

```sql
  query_embedding vector(1536),
  match_threshold float,
  match_count int,
  project_filter uuid
)
RETURNS TABLE (
  id uuid,
  title text,
  description text,
  similarity float
)
LANGUAGE sql STABLE
AS $$
  SELECT
    features.id,
    features.title,
    features.description,
    1 - (features.embedding <=> query_embedding) as similarity
  FROM features
  WHERE 
    features.project_id = project_filter
    AND features.embedding IS NOT NULL
    AND 1 - (features.embedding <=> query_embedding) > match_threshold
  ORDER BY features.embedding <=> query_embedding
  LIMIT match_count;
$$;
```

```typescript
import express from 'express';
import { processWebhook } from './bot/processor';
import crypto from 'crypto';

const app = express();
app.use(express.json());

// Validate Neynar webhook signature
function validateWebhook(signature: string, body: any): boolean {
  const hash = crypto
    .createHmac('sha256', process.env.WEBHOOK_SECRET!)
    .update(JSON.stringify(body))
    .digest('hex');
  return signature === hash;
}

app.get('/health', (req, res) => {
  res.json({ status: 'ok', service: 'roadmapr-bot' });
});

app.post('/webhook/mention', async (req, res) => {
  try {
    const signature = req.headers['x-neynar-signature'] as string;
    
    if (!validateWebhook(signature, req.body)) {
      return res.status(401).json({ error: 'Invalid signature' });
    }

    // Process async
    processWebhook(req.body).catch(err => {
      console.error('Webhook processing error:', err);
    });
    
    res.json({ success: true });
  } catch (err) {
    console.error('Webhook error:', err);
    res.status(500).json({ error: 'Processing failed' });
  }
});

const PORT = process.env.PORT || 3000;
app.listen(PORT, () => {
  console.log(`Bot listening on port ${PORT}`);
});
```

### **src/bot/processor.ts** - Main logic

```typescript
import { extractFeatures } from './extractor';
import { autoTag } from './tagger';
import { findSimilarFeatures } from './similarity';
import { detectProjects } from './router';
import { getCast, getCastThread, postReply, postStandaloneCast, getNeynarScore } from '../neynar/client';
import { 
  createFeature,
  addFeatureSource,
  getProjectByHandle,
  createProject,
  updateFeatureDescription,
  logBotMention,
  checkRateLimited
} from '../db';

export async function processWebhook(data: any) {
  const { cast_hash, author_fid, parent_hash } = data;
  
  // Check already processed
  const existing = await checkProcessed(cast_hash);
  if (existing) {
    console.log(`Already processed: ${cast_hash}`);
    return;
  }

  // Rate limit
  if (await checkRateLimited(author_fid)) {
    await postReply(cast_hash, "Daily limit reached (20/day). Try tomorrow!");
    return;
  }

  // Neynar score check (anti-spam)
  const score = await getNeynarScore(author_fid);
  if (score < parseFloat(process.env.MIN_NEYNAR_SCORE || '0.1')) {
    console.log(`Low score: ${author_fid}`);
    return;
  }

  // Need parent cast
  if (!parent_hash) {
    await postReply(cast_hash, 
      "Reply to a cast with feedback to add it to a roadmap!"
    );
    return;
  }

  // Get parent + thread context
  const parentCast = await getCast(parent_hash);
  if (!parentCast) {
    await postReply(cast_hash, "Couldn't find that cast.");
    return;
  }

  const thread = await getCastThread(parent_hash);
  const fullContext = [
    parentCast.text,
    ...thread.map(c => c.text)
  ].join('\n\n---\n\n');

  // Detect projects mentioned
  const detectedProjects = await detectProjects(fullContext, parentCast);
  
  if (detectedProjects.length === 0) {
    await postReply(cast_hash,
      "Which project is this about? Mention the project handle (e.g., @base) or say 'for @project'"
    );
    await logBotMention(cast_hash, author_fid, parent_hash, {
      error: 'No projects detected'
    });
    return;
  }

  // Check which projects exist
  const projects = [];
  const newProjects = [];
  
  for (const handle of detectedProjects) {
    const project = await getProjectByHandle(handle);
    if (project) {
      projects.push(project);
    } else {
      newProjects.push(handle);
    }
  }

  // If new projects, ask for details
  if (newProjects.length > 0) {
    const mentions = newProjects.map(h => `@${h}`).join(' ');
    await postReply(cast_hash,
      `${mentions} New project detected!\n\nTo create your board, reply with:\n• Project owner FID\n• Token contract address (or "clanker" for default)\n• Short bio\n\nExample: "Owner: 12345, Token: clanker, Bio: Building the future"`
    );
    await logBotMention(cast_hash, author_fid, parent_hash, {
      detected_projects: detectedProjects,
      error: 'Awaiting project setup'
    });
    return;
  }

  // Extract features
  const extracted = await extractFeatures(fullContext);
  
  if (extracted.length === 0) {
    await postReply(cast_hash,
      "Couldn't find actionable feedback. Try describing specific bugs or features."
    );
    await logBotMention(cast_hash, author_fid, parent_hash, {
      detected_projects: detectedProjects,
      features_created: 0
    });
    return;
  }

  // Process each feature
  const results = {
    created: [],
    merged: []
  };

  for (const feature of extracted.slice(0, parseInt(process.env.MAX_FEATURES_PER_CAST || '5'))) {
    for (const project of projects) {
      // Auto-tag
      const tags = await autoTag(feature.title, feature.description);
      
      // Similarity search
      const similar = await findSimilarFeatures(
        project.id,
        feature.title,
        feature.description
      );

      if (similar.length > 0 && similar[0].similarity > 0.85) {
        // MERGE into existing
        await addFeatureSource(similar[0].id, {
          source_cast_hash: parent_hash,
          source_cast_author_fid: parentCast.author.fid,
          source_cast_text: parentCast.text
        });

        // Update description if new one is more detailed
        if (feature.description.length > similar[0].description.length * 0.5) {
          const updated = `${similar[0].description}\n\n---\n\n${feature.description}`;
          await updateFeatureDescription(similar[0].id, updated);
        }

        results.merged.push({
          id: similar[0].id,
          title: similar[0].title,
          project: project.name
        });
      } else {
        // CREATE new feature
        const featureId = await createFeature({
          project_id: project.id,
          title: feature.title,
          description: feature.description,
          submitter_fid: parentCast.author.fid,
          source_cast_hash: parent_hash,
          source_cast_author_fid: parentCast.author.fid,
          tags: tags
        });

        // Add source
        await addFeatureSource(featureId, {
          source_cast_hash: parent_hash,
          source_cast_author_fid: parentCast.author.fid,
          source_cast_text: parentCast.text
        });

        // Create sub-items if present
        if (feature.subItems?.length > 0) {
          for (const sub of feature.subItems) {
            await createFeature({
              project_id: project.id,
              title: sub.title,
              description: sub.description,
              submitter_fid: parentCast.author.fid,
              parent_feature_id: featureId,
              is_sub_item: true
            });
          }
        }

        results.created.push({
          id: featureId,
          title: feature.title,
          project: project.name,
          subItems: feature.subItems?.length || 0
        });
      }
    }
  }

  // Log
  await logBotMention(cast_hash, author_fid, parent_hash, {
    detected_projects: detectedProjects,
    features_created: results.created.length,
    features_merged: results.merged.length
  });

  // Reply to cast
  await postReply(cast_hash, formatReply(results));

  // Standalone cast (quote parent, tag author)
  if (results.created.length > 0) {
    await postStandaloneCast(
      formatStandaloneCast(results.created[0], parentCast),
      parent_hash
    );
  }
}

function formatReply(results: any): string {
  const { created, merged } = results;
  let text = '';
  
  if (created.length > 0) {
    text += `✅ Added ${created.length} to roadmap:\n`;
    created.forEach((f: any) => {
      text += `• ${f.title}`;
      if (f.subItems > 0) text += ` (+${f.subItems} options)`;
      text += `\n`;
    });
  }
  
  if (merged.length > 0) {
    text += `\n🔗 Merged into existing:\n`;
    merged.forEach((m: any) => {
      text += `• ${m.title}\n`;
    });
  }
  
  text += '\nVote: roadmapr.xyz';
  return text;
}

function formatStandaloneCast(feature: any, parentCast: any): string {
  return `📋 New on ${feature.project} roadmap:\n\n"${feature.title}"\n\n` +
         `Suggested by @${parentCast.author.username}\n` +
         `Vote: roadmapr.xyz/features/${feature.id}`;
}

async function checkProcessed(castHash: string): Promise<boolean> {
  const { data } = await supabase
    .from('bot_mentions')
    .select('id')
    .eq('cast_hash', castHash)
    .single();
  return !!data;
}
```

```typescript
import Anthropic from '@anthropic-ai/sdk';

const client = new Anthropic({
  apiKey: process.env.ANTHROPIC_API_KEY,
});

interface ExtractedFeature {
  title: string;
  description: string;
  subItems?: { title: string; description: string }[];
}

export async function extractFeatures(text: string): Promise<ExtractedFeature[]> {
  const prompt = `Extract feature requests or bugs from this feedback.

Rules:
- Extract discrete, actionable items
- If multiple implementation approaches mentioned, create subItems
- Ignore spam, insults, off-topic
- Title: clear, <100 chars
- Description: 1-3 sentences
- Return valid JSON array only

Example:
[
  {
    "title": "Add dark mode",
    "description": "Users want a dark theme for the app",
    "subItems": [
      {"title": "Auto-switch at sunset", "description": "Automatically switch to dark mode in evening"},
      {"title": "OLED black option", "description": "Pure black for OLED screens"}
    ]
  }
]

Input:
${text}

Return JSON array (or empty array):`;

  try {
    const response = await client.messages.create({
      model: 'claude-sonnet-4-20250514',
      max_tokens: 2000,
      messages: [{ role: 'user', content: prompt }],
    });

    const content = response.content[0];
    if (content.type !== 'text') return [];

    const parsed = JSON.parse(content.text);
    return Array.isArray(parsed) ? parsed : [];
  } catch (err) {
    console.error('Extraction error:', err);
    return [];
  }
}
```

### **src/bot/tagger.ts** - Auto-tagging

```typescript
import Anthropic from '@anthropic-ai/sdk';
import { getTagByName, createTag } from '../db/tags';

const client = new Anthropic({
  apiKey: process.env.ANTHROPIC_API_KEY,
});

const PREDEFINED = [
  'bug', 'feature', 'enhancement', 'marketing', 'strategy',
  'design', 'mobile', 'web', 'api', 'documentation', 'performance', 'security'
];

export async function autoTag(title: string, description: string): Promise<string[]> {
  const prompt = `Categorize this with 2-4 relevant tags.

Predefined tags: ${PREDEFINED.join(', ')}

You can suggest new tags (e.g., "notifications", "ux", "onboarding").

Title: ${title}
Description: ${description}

Return JSON array of lowercase tag names:`;

  try {
    const response = await client.messages.create({
      model: 'claude-sonnet-4-20250514',
      max_tokens: 200,
      messages: [{ role: 'user', content: prompt }],
    });

    const content = response.content[0];
    if (content.type !== 'text') return [];

    const parsed = JSON.parse(content.text);
    const tagNames = Array.isArray(parsed) ? parsed : [];
    
    // Ensure tags exist
    const tagIds = [];
    for (const name of tagNames) {
      let tag = await getTagByName(name.toLowerCase());
      if (!tag) {
        tag = await createTag(name.toLowerCase(), 'custom');
      }
      tagIds.push(tag.id);
    }
    
    return tagIds;
  } catch (err) {
    console.error('Tagging error:', err);
    return [];
  }
}
```

```typescript
import OpenAI from 'openai';
import { supabase } from '../db/client';

const openai = new OpenAI({
  apiKey: process.env.OPENAI_API_KEY,
});

export async function findSimilarFeatures(
  projectId: string,
  title: string,
  description: string
): Promise<any[]> {
  // Generate embedding
  const embedding = await generateEmbedding(title + ' ' + description);
  
  // Vector search
  const { data, error } = await supabase.rpc('match_features', {
    query_embedding: embedding,
    match_threshold: parseFloat(process.env.SIMILARITY_THRESHOLD || '0.85'),
    match_count: 3,
    project_filter: projectId
  });

  if (error) {
    console.error('Similarity search error:', error);
    return [];
  }

  return data || [];
}

async function generateEmbedding(text: string): Promise<number[]> {
  const response = await openai.embeddings.create({
    model: 'text-embedding-ada-002',
    input: text,
  });
  
  return response.data[0].embedding;
}

// Store embedding when creating feature
export async function storeFeatureEmbedding(featureId: string, title: string, desc: string) {
  const embedding = await generateEmbedding(title + ' ' + desc);
  
  await supabase
    .from('features')
    .update({ embedding })
    .eq('id', featureId);
}
```

### **src/bot/router.ts** - Project detection

```typescript
import { getCast } from '../neynar/client';

export async function detectProjects(text: string, cast: any): Promise<string[]> {
  const projects = new Set<string>();
  
  // Method 1: @mentions
  const mentions = text.match(/@(\w+)/g) || [];
  for (const mention of mentions) {
    const handle = mention.slice(1).toLowerCase();
    // Filter out common non-project mentions
    if (handle !== 'roadmapr' && handle.length > 2) {
      projects.add(handle);
    }
  }
  
  // Method 2: "for @project" or "on @project"
  const patterns = [
    /(?:for|on|about|regarding)\s+@(\w+)/gi,
    /@(\w+)\s+(?:needs|should|must)/gi
  ];
  
  for (const pattern of patterns) {
    const matches = [...text.matchAll(pattern)];
    matches.forEach(m => projects.add(m[1].toLowerCase()));
  }
  
  // Method 3: Channel context
  if (cast.channel?.id) {
    projects.add(cast.channel.id);
  }
  
  return Array.from(projects);
}
```

### **src/neynar/client.ts** - Farcaster API wrapper

```typescript
import { NeynarAPIClient } from '@neynar/nodejs-sdk';

const client = new NeynarAPIClient(process.env.NEYNAR_API_KEY!);

export async function getCast(hash: string) {
  try {
    const result = await client.lookupCastByHashOrWarpcastUrl(hash, 'hash');
    return result.cast;
  } catch (err) {
    console.error('Get cast error:', err);
    return null;
  }
}

export async function getCastThread(parentHash: string): Promise<any[]> {
  try {
    const result = await client.fetchAllCastsInThread(parentHash);
    return result.result.casts.slice(1); // Exclude parent
  } catch (err) {
    console.error('Get thread error:', err);
    return [];
  }
}

export async function postReply(replyToHash: string, text: string) {
  try {
    await client.publishCast(
      process.env.NEYNAR_BOT_SIGNER_UUID!,
      text,
      { replyTo: replyToHash }
    );
  } catch (err) {
    console.error('Post reply error:', err);
  }
}

export async function postStandaloneCast(text: string, embedHash?: string) {
  try {
    const options: any = {};
    if (embedHash) {
      options.embeds = [{ castId: { hash: embedHash } }];
    }
    
    await client.publishCast(
      process.env.NEYNAR_BOT_SIGNER_UUID!,
      text,
      options
    );
  } catch (err) {
    console.error('Post cast error:', err);
  }
}

export async function getNeynarScore(fid: number): Promise<number> {
  try {
    const user = await client.fetchBulkUsers([fid]);
    return user.users[0]?.experimental?.neynar_user_score || 0;
  } catch (err) {
    console.error('Get score error:', err);
    return 0;
  }
```

```typescript
// components/TagFilter.tsx

'use client';

import { useState, useEffect } from 'react';
import { supabase } from '@/lib/supabase';

export default function TagFilter({ projectId, onFilterChange }) {
  const [tags, setTags] = useState([]);
  const [selected, setSelected] = useState<string[]>([]);

  useEffect(() => {
    loadTags();
  }, [projectId]);

  async function loadTags() {
    const { data } = await supabase
      .from('tags')
      .select('*')
      .order('name');
    
    setTags(data || []);
  }

  function toggleTag(tagId: string) {
    setSelected(prev => 
      prev.includes(tagId) 
        ? prev.filter(id => id !== tagId)
        : [...prev, tagId]
    );
    onFilterChange?.(selected);
  }

  return (
    <div className="mb-6">
      <div className="flex flex-wrap gap-2">
        {tags.map(tag => (
          <button
            key={tag.id}
            onClick={() => toggleTag(tag.id)}
            className={`px-3 py-1 rounded-full text-sm ${
              selected.includes(tag.id)
                ? 'bg-blue-600 text-white'
                : 'bg-gray-100 text-gray-700 hover:bg-gray-200'
            }`}
          >
            {tag.name}
          </button>
        ))}
      </div>
    </div>
  );
}
```

## Token Voting Component

```typescript
// components/VoteButton.tsx

'use client';

import { useState } from 'react';
import { useAccount } from 'wagmi';
import { parseUnits } from 'viem';

export default function VoteButton({ feature, compact = false }) {
  const { address } = useAccount();
  const [amount, setAmount] = useState('10');
  const [voting, setVoting] = useState(false);

  async function handleVote(direction: 'up' | 'down') {
    if (!address) {
      // Prompt wallet connection
      return;
    }

    setVoting(true);

    try {
      // 1. Lock tokens in escrow contract
      const tokenAmount = parseUnits(amount, 18);
      const fee = tokenAmount * BigInt(1) / BigInt(100); // 1% fee
      const total = tokenAmount + fee;

      // Call contract to lock tokens
      await lockTokensForVote({
        featureId: feature.id,
        amount: tokenAmount,
        fee: fee,
        direction
      });

      // 2. Record vote in DB
      await fetch('/api/votes', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          feature_id: feature.id,
          voter_address: address,
          weight: direction === 'up' ? tokenAmount : -tokenAmount,
          tokens_locked: tokenAmount,
          fee_amount: fee
        })
      });

      // Refresh
      window.location.reload();
    } catch (err) {
      console.error('Vote error:', err);
      alert('Vote failed. Try again.');
    } finally {
      setVoting(false);
    }
  }

  return (
    <div className="flex items-center gap-3">
      <input
        type="number"
        value={amount}
        onChange={e => setAmount(e.target.value)}
        className="w-24 px-3 py-2 border rounded-lg"
        placeholder="Amount"
      />
      
      <button
        onClick={() => handleVote('up')}
        disabled={voting}
        className="px-4 py-2 bg-green-600 text-white rounded-lg hover:bg-green-700 disabled:opacity-50"
      >
        ↑ Upvote
      </button>
      
      <button
        onClick={() => handleVote('down')}
        disabled={voting}
        className="px-4 py-2 bg-red-600 text-white rounded-lg hover:bg-red-700 disabled:opacity-50"
      >
        ↓ Downvote
      </button>
      
      <span className="text-sm text-gray-600">
        +1% fee ({(parseFloat(amount) * 0.01).toFixed(2)} tokens)
      </span>
    </div>
  );
}
```
